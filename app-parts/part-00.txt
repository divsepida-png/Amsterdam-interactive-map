  function esc(value=''){
    return String(value).replace(/[&<>"']/g,char=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[char]));

  function clamp(value,min,max){return Math.min(max,Math.max(min,value))}
  function radians(value){return value*Math.PI/180}
  function haversine(a,b){
    const R=6371000,dLat=radians(b.lat-a.lat),dLon=radians(b.lon-a.lon);
    const h=Math.sin(dLat/2)**2+Math.cos(radians(a.lat))*Math.cos(radians(b.lat))*Math.sin(dLon/2)**2;
    return 2*R*Math.asin(Math.min(1,Math.sqrt(h)));
  }
  function formatDistance(metres){

  function rectanglePolygon(lon,lat,widthM=18,heightM=12){
    const dx=(widthM/2)/(111320*Math.cos(radians(lat)));
    const dy=(heightM/2)/110540;
    return [[[lon-dx,lat-dy],[lon+dx,lat-dy],[lon+dx,lat+dy],[lon-dx,lat+dy],[lon-dx,lat-dy]]];
  }
  function circlePolygon(lon,lat,radiusM=30,steps=28){
    const coords=[];
    for(let i=0;i<=steps;i++){
      const angle=(i/steps)*Math.PI*2;
      const dx=(Math.cos(angle)*radiusM)/(111320*Math.cos(radians(lat)));
      const dy=(Math.sin(angle)*radiusM)/110540;
      coords.push([lon+dx,lat+dy]);
    }
    return [coords];
  }

  function categoryLabel(item){

  }
  function makeFrontageBlocks(frontages,kind,colour){
    const features=[];
    let id=0;
    for(const frontage of frontages){
      const [a,b]=frontage.points;
      const distance=haversine({lat:a[0],lon:a[1]},{lat:b[0],lon:b[1]});
      const count=Math.max(1,Math.ceil(distance/18));
      for(let i=0;i<count;i++){
        const t=(i+.5)/count;
        const lat=a[0]+(b[0]-a[0])*t;
        const lon=a[1]+(b[1]-a[1])*t;
        const item={id:`${kind}-${id++}`,name:frontage.name,kind,lat,lon,symbol:kind==='red-light'?'R':'B',note:'Approximate public street-frontage block; active individual windows can change.'};
        features.push(polygonFeature(item,12,8,{colour}));
      }
    }
    return features;
  }
